using System.Collections.Generic;
using System.IO;
using System.Linq;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

public class CreasePatternEditor : MonoBehaviour
{
    [Header("编辑器设置")]
    public Camera editorCamera;
    public float gridSize = 1f;
    public float snapDistance = 0.25f;
    public Material vertexMaterial;
    public Material creaseMaterial;
    private Material faceSharedMaterial;
    public float vertexSize = 0.1f;
    public float creaseWidth = 0.05f;

    [Header("UI 元素")]
    public Button addVertexButton;
    public Button addCreaseButton;
    public Button moveVertexButton;
    public Button markfacesButton;
    public Button saveButton;
    public Button clearButton;
    [Header("折痕类型选择（三选一）")]
    public ToggleGroup creaseTypeGroup;
    public Toggle mountainToggle;
    public Toggle valleyToggle;
    public Toggle boundaryToggle;
    public TMP_InputField fileNameInput;

    private enum EditorMode { None, AddVertex, AddCrease, MoveVertex, MarkFaces }
    private EditorMode currentMode = EditorMode.None;

    private List<OrigamiVertex> vertices = new List<OrigamiVertex>();
    private List<OrigamiCrease> creases = new List<OrigamiCrease>();
    private List<OrigamiFace> faces = new List<OrigamiFace>();
    private List<GameObject> vertexObjects = new List<GameObject>();
    private List<GameObject> creaseObjects = new List<GameObject>();
    private List<GameObject> faceObjects = new List<GameObject>();

    private int selectedVertexIndex = -1;
    private int startVertexIndex = -1;

    // 新增：半边数据结构
    private class HalfEdge
    {
        public int vertexIndex;       // 起点顶点索引
        public HalfEdge next;         // 下一条半边
        public HalfEdge twin;         // 对边（反向半边）
        public int faceIndex = -1;    // 所属面索引（-1表示未分配）
        public Vector3 position;      // 用于射线检测的3D位置
    }

    private Dictionary<int, HalfEdge> halfEdges = new Dictionary<int, HalfEdge>();
    private List<List<int>> faceVertexLists = new List<List<int>>(); // 缓存所有面的顶点列表

    // 预生成材质
    private Material mountainMat;
    private Material valleyMat;
    private Material boundaryMat;

    void Start()
    {
        if (editorCamera == null)
            editorCamera = Camera.main;

        // 绑定UI事件
        if (addVertexButton != null) addVertexButton.onClick.AddListener(() => SetMode(EditorMode.AddVertex));
        if (addCreaseButton != null) addCreaseButton.onClick.AddListener(() => SetMode(EditorMode.AddCrease));
        if (moveVertexButton != null) moveVertexButton.onClick.AddListener(() => SetMode(EditorMode.MoveVertex));
        if (markfacesButton != null) markfacesButton.onClick.AddListener(() => SetMode(EditorMode.MarkFaces));
        if (saveButton != null) saveButton.onClick.AddListener(SaveToJson);
        if (clearButton != null) clearButton.onClick.AddListener(ClearAll);

        // 初始化折痕材质
        mountainMat = new Material(Shader.Find("Unlit/Color")) { color = Color.red };
        valleyMat = new Material(Shader.Find("Unlit/Color")) { color = Color.blue };
        boundaryMat = new Material(Shader.Find("Unlit/Color")) { color = Color.black };

        // ✅ 修复：使用支持透明的Shader
        faceSharedMaterial = new Material(Shader.Find("Unlit/Transparent"));
        faceSharedMaterial.color = new Color(0f, 1f, 0f, 0.35f);
        faceSharedMaterial.SetInt("_SrcBlend", (int)UnityEngine.Rendering.BlendMode.SrcAlpha);
        faceSharedMaterial.SetInt("_DstBlend", (int)UnityEngine.Rendering.BlendMode.OneMinusSrcAlpha);
        faceSharedMaterial.SetInt("_ZWrite", 0);
        faceSharedMaterial.DisableKeyword("_ALPHATEST_ON");
        faceSharedMaterial.EnableKeyword("_ALPHABLEND_ON");
        faceSharedMaterial.DisableKeyword("_ALPHAPREMULTIPLY_ON");
        faceSharedMaterial.renderQueue = 3000;
    }

    void Update()
    {
        // 限制仅在编辑模式处理输入
        if (currentMode == EditorMode.None) return;

        if (Input.GetMouseButtonDown(0))
        {
            Vector3 worldPos = GetMouseWorldPosition();
            HandleClick(worldPos);
        }

        // 顶点移动
        if (currentMode == EditorMode.MoveVertex && selectedVertexIndex >= 0)
        {
            if (Input.GetMouseButton(0))
                MoveSelectedVertex(GetMouseWorldPosition());
            else if (Input.GetMouseButtonUp(0))
                selectedVertexIndex = -1;
        }

        // 退出当前操作
        if (Input.GetKeyDown(KeyCode.Escape))
            CancelCurrentOperation();
    }

    private void HandleClick(Vector3 worldPos)
    {
        switch (currentMode)
        {
            case EditorMode.AddVertex: AddVertex(worldPos); break;
            case EditorMode.AddCrease: HandleCreaseCreation(worldPos); break;
            case EditorMode.MoveVertex: SelectVertex(worldPos); break;
            case EditorMode.MarkFaces: MarkFaceAt(worldPos); break;
        }
    }

    private void SetMode(EditorMode mode)
    {
        currentMode = mode;
        CancelCurrentOperation();

        // 更新按钮状态
        UpdateButtonInteractability();

        Debug.Log($"当前模式: {mode}");
    }

    private void UpdateButtonInteractability()
    {
        if (addVertexButton != null) addVertexButton.interactable = (currentMode != EditorMode.AddVertex);
        if (addCreaseButton != null) addCreaseButton.interactable = (currentMode != EditorMode.AddCrease);
        if (moveVertexButton != null) moveVertexButton.interactable = (currentMode != EditorMode.MoveVertex);
        if (markfacesButton != null) markfacesButton.interactable = (currentMode != EditorMode.MarkFaces);
    }

    private Vector3 GetMouseWorldPosition()
    {
        Ray ray = editorCamera.ScreenPointToRay(Input.mousePosition);
        Plane plane = new Plane(Vector3.up, Vector3.zero);

        if (plane.Raycast(ray, out float distance))
        {
            Vector3 worldPos = ray.GetPoint(distance);
            worldPos.y = 0;
            if (gridSize > 0)
            {
                worldPos.x = Mathf.Round(worldPos.x / gridSize) * gridSize;
                worldPos.z = Mathf.Round(worldPos.z / gridSize) * gridSize;
            }
            return worldPos;
        }
        return Vector3.zero;
    }

    private void AddVertex(Vector3 position)
    {
        position.y = 0;
        int existingIndex = FindNearestVertex(position);
        if (existingIndex >= 0 && Vector3.Distance(position, new Vector3(vertices[existingIndex].x, 0, vertices[existingIndex].z)) < snapDistance)
        {
            Debug.Log("已存在相近顶点，不创建新顶点");
            return;
        }

        // 创建新顶点
        OrigamiVertex newVertex = new OrigamiVertex { x = position.x, y = 0, z = position.z };
        vertices.Add(newVertex);

        // 创建可视化对象
        GameObject vertexObj = GameObject.CreatePrimitive(PrimitiveType.Sphere);
        vertexObj.transform.position = position;
        vertexObj.transform.localScale = Vector3.one * vertexSize;
        vertexObj.transform.parent = transform;
        vertexObj.name = $"Vertex_{vertices.Count}";
        if (vertexMaterial != null)
            vertexObj.GetComponent<Renderer>().material = vertexMaterial;
        vertexObjects.Add(vertexObj);

        Debug.Log($"添加顶点 #{vertices.Count} 在 ({position.x}, 0, {position.z})");
    }

    private void HandleCreaseCreation(Vector3 position)
    {
        int vertexIndex = FindNearestVertex(position);
        if (vertexIndex < 0)
        {
            Debug.Log("未找到顶点，无法创建折痕");
            return;
        }

        if (startVertexIndex < 0)
        {
            // 选择起始顶点
            startVertexIndex = vertexIndex;
            Debug.Log($"选择起始顶点 #{startVertexIndex + 1}");
            vertexObjects[startVertexIndex].GetComponent<Renderer>().material.color = Color.yellow;
        }
        else
        {
            // 创建折痕
            if (startVertexIndex != vertexIndex)
                CreateCrease(startVertexIndex, vertexIndex);

            // 重置起始顶点
            vertexObjects[startVertexIndex].GetComponent<Renderer>().material = vertexMaterial;
            startVertexIndex = -1;
        }
    }

    private void CreateCrease(int v1Index, int v2Index)
    {
        // 检查重复折痕
        foreach (var c in creases)
        {
            if ((c.v1 == v1Index + 1 && c.v2 == v2Index + 1) ||
                (c.v1 == v2Index + 1 && c.v2 == v1Index + 1))
            {
                Debug.Log("已存在相同的折痕");
                return;
            }
        }

        // 获取折痕类型
        bool isMountain = mountainToggle != null && mountainToggle.isOn;
        bool isValley = valleyToggle != null && valleyToggle.isOn;
        bool isBoundary = boundaryToggle != null && boundaryToggle.isOn;

        // 创建新折痕
        OrigamiCrease newCrease = new OrigamiCrease
        {
            id = creases.Count + 1,
            v1 = v1Index + 1,
            v2 = v2Index + 1,
            restAngle = 0,
            minAngle = -180,
            maxAngle = 180,
            stiffness = 1,
            width = creaseWidth,
            boundary = isBoundary,
            mountain = isMountain
        };
        creases.Add(newCrease);

        // 创建可视化对象
        GameObject creaseObj = new GameObject($"Crease_{newCrease.id}");
        creaseObj.transform.parent = transform;

        LineRenderer lr = creaseObj.AddComponent<LineRenderer>();
        lr.positionCount = 2;
        lr.SetPosition(0, new Vector3(vertices[v1Index].x, vertices[v1Index].y, vertices[v1Index].z));
        lr.SetPosition(1, new Vector3(vertices[v2Index].x, vertices[v2Index].y, vertices[v2Index].z));

        // 边界折痕加宽
        float lineWidth = isBoundary ? creaseWidth * 1.5f : creaseWidth;
        lr.startWidth = lr.endWidth = lineWidth;

        // 设置材质
        lr.material = isBoundary ? boundaryMat : (isMountain ? mountainMat : valleyMat);
        creaseObjects.Add(creaseObj);

        string creaseType = isBoundary ? "边界" : (isMountain ? "山折" : "谷折");
        Debug.Log($"创建{creaseType} #{newCrease.id} 连接顶点 #{v1Index + 1} 和 #{v2Index + 1}");
    }

    private void SelectVertex(Vector3 position)
    {
        selectedVertexIndex = FindNearestVertex(position);
        if (selectedVertexIndex >= 0)
            Debug.Log($"选中顶点 #{selectedVertexIndex + 1}");
    }

    private void MarkFaceAt(Vector3 clickPos)
    {
        // 1. 构建/更新半边数据结构
        if (!RebuildHalfEdgeStructure())
        {
            Debug.LogWarning("⚠️ 无法构建面拓扑（检查折痕是否形成封闭区域）");
            return;
        }

        // 2. 找到最近的半边
        HalfEdge nearestEdge = FindNearestHalfEdge(clickPos);
        if (nearestEdge == null)
        {
            Debug.LogWarning("⚠️ 未找到附近折痕，无法标记面");
            return;
        }

        // 3. 获取半边所属的面
        int faceIndex = GetFaceIndex(nearestEdge);
        if (faceIndex < 0)
        {
            Debug.LogWarning("⚠️ 未找到有效面（折痕可能未闭合）");
            return;
        }

        // 4. 获取面的顶点列表（1-based索引）
        List<int> faceVertices = faceVertexLists[faceIndex];
        if (faceVertices.Count < 3)
        {
            Debug.LogWarning("⚠️ 无效面（顶点数不足）");
            return;
        }

        // 5. 检查重复面
        if (IsFaceDuplicate(faceVertices))
        {
            Debug.LogWarning("⚠️ 该面已存在，跳过创建");
            return;
        }

        // 6. 创建新面
        var newFace = new OrigamiFace
        {
            id = faces.Count + 1,
            vertices = new List<int>(faceVertices),
            rigid = true
        };
        faces.Add(newFace);

        // 7. 可视化
        VisualizeFace(faceVertices.Select(v => v - 1).ToList());

        Debug.Log($"✅ 已标记新面 #{newFace.id}（顶点数: {faceVertices.Count}）");
    }

    private bool RebuildHalfEdgeStructure()
    {
        // 清空现有结构
        halfEdges.Clear();
        faceVertexLists.Clear();

        // 1. 为每条折痕创建两条半边
        foreach (var crease in creases)
        {
            int v1 = crease.v1 - 1;
            int v2 = crease.v2 - 1;

            // 创建正向半边
            HalfEdge he1 = new HalfEdge { vertexIndex = v1 };
            HalfEdge he2 = new HalfEdge { vertexIndex = v2 };

            // 设置对边关系
            he1.twin = he2;
            he2.twin = he1;

            // 存储到字典（使用唯一键）
            halfEdges[GetHalfEdgeKey(v1, v2)] = he1;
            halfEdges[GetHalfEdgeKey(v2, v1)] = he2;
        }

        // 2. 为每个顶点构建出边列表（按角度排序）
        var vertexOutEdges = new Dictionary<int, List<HalfEdge>>();

        foreach (var kvp in halfEdges)
        {
            int startVertex = kvp.Value.vertexIndex;
            if (!vertexOutEdges.ContainsKey(startVertex))
                vertexOutEdges[startVertex] = new List<HalfEdge>();

            vertexOutEdges[startVertex].Add(kvp.Value);
        }

        // 3. 为每个顶点的出边按角度排序
        foreach (var kvp in vertexOutEdges)
        {
            int vertexIdx = kvp.Key;
            Vector3 vertexPos = GetVertexPosition(vertexIdx);

            kvp.Value.Sort((a, b) =>
            {
                Vector3 dirA = (GetVertexPosition(a.twin.vertexIndex) - vertexPos).normalized;
                Vector3 dirB = (GetVertexPosition(b.twin.vertexIndex) - vertexPos).normalized;

                // 计算相对于X轴的角度（XZ平面）
                float angleA = Mathf.Atan2(dirA.z, dirA.x);
                float angleB = Mathf.Atan2(dirB.z, dirB.x);

                return angleA.CompareTo(angleB);
            });
        }

        // 4. 建立半边连接关系（next指针）
        foreach (var kvp in vertexOutEdges)
        {
            List<HalfEdge> edges = kvp.Value;
            for (int i = 0; i < edges.Count; i++)
            {
                HalfEdge current = edges[i];
                HalfEdge nextEdge = edges[(i + 1) % edges.Count];
                current.next = nextEdge.twin; // 下一个半边是相邻边的对边
            }
        }

        // 5. 遍历所有半边，构建面
        int currentFaceIndex = 0;

        foreach (var he in halfEdges.Values)
        {
            if (he.faceIndex >= 0) continue; // 已分配面

            // 从当前半边开始遍历
            List<int> faceVertices = new List<int>();
            HalfEdge current = he;

            do
            {
                // 记录顶点（1-based索引）
                faceVertices.Add(current.vertexIndex + 1);
                current.faceIndex = currentFaceIndex;

                current = current.next;
            }
            while (current != he && current != null && faceVertices.Count < 100); // 安全限制

            // 验证有效面
            if (current != he || faceVertices.Count < 3)
            {
                // 无效面：重置faceIndex
                current = he;
                do
                {
                    current.faceIndex = -1;
                    current = current.next;
                } while (current != he && current != null);
                continue;
            }

            // 添加到面列表
            faceVertexLists.Add(faceVertices);
            currentFaceIndex++;
        }

        return faceVertexLists.Count > 0;
    }

    private int GetHalfEdgeKey(int start, int end)
    {
        return (start << 16) | end;
    }

    private HalfEdge FindNearestHalfEdge(Vector3 clickPos)
    {
        float minDist = float.MaxValue;
        HalfEdge nearest = null;

        // 遍历所有半边（取唯一折痕）
        foreach (var kvp in halfEdges)
        {
            // 每条折痕只处理一次
            if (kvp.Key % 2 != 0) continue;

            HalfEdge he = kvp.Value;
            Vector3 start = GetVertexPosition(he.vertexIndex);
            Vector3 end = GetVertexPosition(he.twin.vertexIndex);

            // 计算点到线段的距离
            Vector3 closest = ClosestPointOnSegment(start, end, clickPos);
            float dist = Vector3.Distance(clickPos, closest);

            if (dist < minDist)
            {
                minDist = dist;
                nearest = he;
            }
        }

        return minDist < 5f ? nearest : null; // 5米内有效
    }

    private Vector3 ClosestPointOnSegment(Vector3 a, Vector3 b, Vector3 p)
    {
        Vector3 ab = b - a;
        float t = Vector3.Dot(p - a, ab) / Vector3.Dot(ab, ab);
        t = Mathf.Clamp01(t);
        return a + t * ab;
    }

    private int GetFaceIndex(HalfEdge edge)
    {
        // 检查半边是否已分配面
        if (edge.faceIndex >= 0)
            return edge.faceIndex;

        // 尝试通过相邻半边获取
        if (edge.twin != null && edge.twin.faceIndex >= 0)
            return edge.twin.faceIndex;

        return -1;
    }

    private bool IsFaceDuplicate(List<int> newVertices)
    {
        if (newVertices.Count == 0) return true;

        var newSet = new HashSet<int>(newVertices);
        foreach (var f in faces)
        {
            if (f.vertices.Count == newVertices.Count &&
                new HashSet<int>(f.vertices).SetEquals(newSet))
            {
                return true;
            }
        }
        return false;
    }

    private List<int> FindEnclosedFace(int startVertexIndex, Vector3 clickPosition)
    {
        // 1. 构建顶点邻接表
        var adjacency = BuildVertexAdjacency();

        // 2. 检查起点有效性
        if (!adjacency.TryGetValue(startVertexIndex, out var neighbors) || neighbors.Count < 2)
        {
            Debug.Log($"顶点 {startVertexIndex + 1} 连接度不足 ({neighbors?.Count ?? 0})，无法形成面");
            return null;
        }

        // 3. ⭐ 生成多个候选面（尝试不同起始边和方向）
        List<List<int>> candidateFaces = new List<List<int>>();

        // 尝试每个邻居作为起始方向
        foreach (int firstNeighbor in neighbors)
        {
            // 顺时针路径
            var cwPath = TracePath(startVertexIndex, firstNeighbor, adjacency, preferClockwise: true);
            if (cwPath != null && cwPath.Count >= 3 && IsSimpleCycle(cwPath))
            {
                candidateFaces.Add(cwPath);
            }

            // 逆时针路径
            var ccwPath = TracePath(startVertexIndex, firstNeighbor, adjacency, preferClockwise: false);
            if (ccwPath != null && ccwPath.Count >= 3 && IsSimpleCycle(ccwPath))
            {
                candidateFaces.Add(ccwPath);
            }
        }

        // 4. 去重：移除顶点集相同的面
        candidateFaces = RemoveDuplicateFaces(candidateFaces);

        if (candidateFaces.Count == 0)
        {
            Debug.Log("未找到任何有效候选面");
            return null;
        }

        // 5. ⭐⭐⭐ 筛选包含点击位置的面
        var containingFaces = new List<List<int>>();
        foreach (var face in candidateFaces)
        {
            if (IsPointInFace(clickPosition, face))
            {
                containingFaces.Add(face);
            }
        }

        // 6. 选择最佳面
        List<int> selectedFace = null;

        if (containingFaces.Count > 0)
        {
            // 优先选择包含点击点的面中顶点数最少的
            selectedFace = containingFaces.OrderBy(f => f.Count).First();
            Debug.Log($"✅ 找到 {containingFaces.Count} 个包含点击点的面，选择最小面（{selectedFace.Count}顶点）");
        }
        else
        {
            // 备选：无面包含点击点，选择最小面并警告
            selectedFace = candidateFaces.OrderBy(f => f.Count).First();
            Debug.LogWarning($"⚠️ 未找到包含点击位置的面（共 {candidateFaces.Count} 个候选面），返回最小面（{selectedFace.Count}顶点）");
        }

        return selectedFace;
    }

    // 检查点是否在面内（射线法）
    private bool IsPointInFace(Vector3 point, List<int> faceVertices)
    {
        if (faceVertices.Count < 3) return false;

        // 2D投影（XZ平面）
        Vector2[] polygon = new Vector2[faceVertices.Count];
        for (int i = 0; i < faceVertices.Count; i++)
        {
            Vector3 v = GetVertexPosition(faceVertices[i]);
            polygon[i] = new Vector2(v.x, v.z);
        }

        Vector2 testPoint = new Vector2(point.x, point.z);
        return IsPointInPolygon(polygon, testPoint);
    }

    // 射线法检测点是否在多边形内
    private bool IsPointInPolygon(Vector2[] polygon, Vector2 point)
    {
        bool inside = false;
        int n = polygon.Length;

        for (int i = 0, j = n - 1; i < n; j = i++)
        {
            // 检查点是否在边上（包含边界）
            if (IsPointOnLineSegment(polygon[i], polygon[j], point))
                return true;

            // 射线相交检测
            if (((polygon[i].y > point.y) != (polygon[j].y > point.y)) &&
                (point.x < (polygon[j].x - polygon[i].x) * (point.y - polygon[i].y) /
                (polygon[j].y - polygon[i].y) + polygon[i].x))
            {
                inside = !inside;
            }
        }

        return inside;
    }

    // 检查点是否在线段上
    private bool IsPointOnLineSegment(Vector2 a, Vector2 b, Vector2 p)
    {
        // 检查共线性
        float cross = (b.x - a.x) * (p.y - a.y) - (b.y - a.y) * (p.x - a.x);
        if (Mathf.Abs(cross) > 0.01f) // 允许小误差
            return false;

        // 检查在边界内
        float dot = (p.x - a.x) * (p.x - b.x) + (p.y - a.y) * (p.y - b.y);
        return dot <= 0.01f; // 允许小误差
    }

    // 去重：移除顶点集相同的面
    private List<List<int>> RemoveDuplicateFaces(List<List<int>> faces)
    {
        var uniqueFaces = new List<List<int>>();
        var seenSets = new List<HashSet<int>>();

        foreach (var face in faces)
        {
            var vertexSet = new HashSet<int>(face);

            bool isDuplicate = false;
            foreach (var seen in seenSets)
            {
                if (seen.SetEquals(vertexSet))
                {
                    isDuplicate = true;
                    break;
                }
            }

            if (!isDuplicate)
            {
                uniqueFaces.Add(face);
                seenSets.Add(vertexSet);
            }
        }

        return uniqueFaces;
    }

    // 路径追踪方法
    private List<int> TracePath(int start, int first, Dictionary<int, List<int>> adjacency, bool preferClockwise)
    {
        List<int> path = new List<int> { start, first };
        int prev = start;
        int current = first;

        const int MAX_STEPS = 30;
        for (int step = 0; step < MAX_STEPS; step++)
        {
            int next = FindNextVertexByTurningDirection(current, prev, adjacency, path, preferClockwise);

            // 检查是否闭合
            if (next == start && path.Count >= 3)
            {
                path.Add(next); // 闭合
                if (IsSimpleCycle(path))
                {
                    return RemoveDuplicateStartEnd(new List<int>(path));
                }
            }

            // 无效移动终止
            if (next == -1)
            {
                // 尝试备选策略：允许访问起点（如果接近）
                if (path.Count >= 3 && adjacency[current].Contains(start))
                {
                    var testPath = new List<int>(path) { start };
                    if (IsSimpleCycle(testPath))
                        return RemoveDuplicateStartEnd(testPath);
                }
                return null;
            }

            // 避免无限循环
            if (path.Contains(next) && next != start)
                return null;

            // 更新路径
            path.Add(next);
            prev = current;
            current = next;
        }

        return null; // 超出最大步数
    }

    // 核心：基于转向方向查找下一个顶点
    private int FindNextVertexByTurningDirection(
        int current,
        int prev,
        Dictionary<int, List<int>> adjacency,
        List<int> currentPath,
        bool preferClockwise)
    {
        if (!adjacency.TryGetValue(current, out var candidates))
            return -1;

        Vector3 incomingDir = GetNormalizedDirection(prev, current);
        var validTurns = new List<(int next, float angle, bool isClockwise)>();

        foreach (int next in candidates)
        {
            // 规则1：不能立即回头
            if (next == prev) continue;

            // 规则2：避免自交（除非是起点）
            if (currentPath.Contains(next) && next != currentPath[0]) continue;

            // 计算转向信息
            float turnAngle = CalculateTurningAngle(prev, current, next);
            bool isClockwise = IsClockwiseTurn(prev, current, next);

            // 规则3：过滤无效角度
            if (turnAngle > 170f || turnAngle < 5f) continue; // 避免直线和尖角

            validTurns.Add((next, turnAngle, isClockwise));
        }

        if (validTurns.Count == 0) return -1;

        // 按转向方向和角度排序
        if (preferClockwise)
        {
            // 优先右转(顺时针)，然后是最小转角
            validTurns = validTurns
                .OrderBy(t => t.isClockwise ? 0 : 1) // 顺时针优先
                .ThenBy(t => t.angle)                 // 然后按角度排序
                .ToList();
        }
        else
        {
            // 优先左转(逆时针)，然后是最小转角
            validTurns = validTurns
                .OrderBy(t => t.isClockwise ? 1 : 0) // 逆时针优先
                .ThenBy(t => t.angle)                 // 然后按角度排序
                .ToList();
        }

        return validTurns[0].next;
    }

    // 其他辅助方法保持不变
    private Dictionary<int, List<int>> BuildVertexAdjacency()
    {
        var adjacency = new Dictionary<int, List<int>>();

        foreach (var crease in creases)
        {
            int v1 = crease.v1 - 1;
            int v2 = crease.v2 - 1;

            if (!adjacency.ContainsKey(v1)) adjacency[v1] = new List<int>();
            if (!adjacency.ContainsKey(v2)) adjacency[v2] = new List<int>();

            adjacency[v1].Add(v2);
            adjacency[v2].Add(v1);
        }

        // 移除重复邻居
        foreach (var key in adjacency.Keys.ToList())
        {
            adjacency[key] = adjacency[key].Distinct().ToList();
        }

        return adjacency;
    }

    private int SelectBestStartEdge(int startVertex, Dictionary<int, List<int>> adjacency)
    {
        if (!adjacency.TryGetValue(startVertex, out var neighbors) || neighbors.Count == 0)
            return -1;

        // 选择形成最小角度的两条边中的第一条
        float minAngle = float.MaxValue;
        int bestNeighbor = neighbors[0];

        for (int i = 0; i < neighbors.Count; i++)
        {
            for (int j = i + 1; j < neighbors.Count; j++)
            {
                float angle = CalculateTurningAngle(neighbors[i], startVertex, neighbors[j]);
                if (angle < minAngle)
                {
                    minAngle = angle;
                    bestNeighbor = neighbors[i];
                }
            }
        }

        // 安全保护：避免极小角度
        if (minAngle < 5f)
        {
            foreach (var n in neighbors)
            {
                if (Vector3.Distance(GetVertexPosition(startVertex), GetVertexPosition(n)) > 0.1f)
                    return n;
            }
        }

        return bestNeighbor;
    }

    private Vector3 GetNormalizedDirection(int fromIdx, int toIdx)
    {
        Vector3 from = GetVertexPosition(fromIdx);
        Vector3 to = GetVertexPosition(toIdx);
        Vector3 dir = to - from;
        dir.y = 0; // 忽略高度，假设平面
        return dir.normalized;
    }

    private Vector3 GetVertexPosition(int vertexIndex)
    {
        if (vertexIndex < 0 || vertexIndex >= vertices.Count)
            return Vector3.zero;

        var v = vertices[vertexIndex];
        return new Vector3(v.x, v.y, v.z);
    }

    private float CalculateTurningAngle(int prev, int current, int next)
    {
        Vector3 incoming = GetNormalizedDirection(prev, current);
        Vector3 outgoing = GetNormalizedDirection(current, next);
        return Vector3.Angle(incoming, outgoing);
    }

    private bool IsClockwiseTurn(int prev, int current, int next)
    {
        Vector3 a = GetVertexPosition(prev);
        Vector3 b = GetVertexPosition(current);
        Vector3 c = GetVertexPosition(next);

        // 2D投影（XZ平面）
        Vector2 a2 = new Vector2(a.x, a.z);
        Vector2 b2 = new Vector2(b.x, b.z);
        Vector2 c2 = new Vector2(c.x, c.z);

        // 计算叉积
        Vector2 ab = b2 - a2;
        Vector2 bc = c2 - b2;
        float cross = ab.x * bc.y - ab.y * bc.x;

        return cross < 0; // 负值 = 顺时针（右转）
    }

    private List<int> RemoveDuplicateStartEnd(List<int> path)
    {
        if (path.Count >= 2 && path[0] == path[^1])
        {
            path.RemoveAt(path.Count - 1);
        }
        return path;
    }

    // 验证简单环（无自交）
    private bool IsSimpleCycle(List<int> cycle)
    {
        if (cycle.Count < 3) return false;

        // 检查重复顶点（除首尾）
        var vertexSet = new HashSet<int>();
        for (int i = 0; i < cycle.Count - 1; i++)
        {
            if (vertexSet.Contains(cycle[i]))
                return false;
            vertexSet.Add(cycle[i]);
        }

        // 检查边相交
        for (int i = 0; i < cycle.Count - 1; i++)
        {
            Vector3 a1 = GetVertexPosition(cycle[i]);
            Vector3 a2 = GetVertexPosition(cycle[i + 1]);

            for (int j = i + 2; j < cycle.Count - 1; j++)
            {
                // 跳过相邻边
                if (j == i || j == i + 1 || (i == 0 && j == cycle.Count - 2))
                    continue;

                Vector3 b1 = GetVertexPosition(cycle[j]);
                Vector3 b2 = GetVertexPosition(cycle[j + 1]);

                if (DoSegmentsIntersect(a1, a2, b1, b2))
                    return false;
            }
        }

        return true;
    }

    // 2D线段相交检测
    private bool DoSegmentsIntersect(Vector3 p1, Vector3 p2, Vector3 p3, Vector3 p4)
    {
        Vector2 a = new Vector2(p1.x, p1.z);
        Vector2 b = new Vector2(p2.x, p2.z);
        Vector2 c = new Vector2(p3.x, p3.z);
        Vector2 d = new Vector2(p4.x, p4.z);

        float Cross(Vector2 o, Vector2 a, Vector2 b)
            => (a.x - o.x) * (b.y - o.y) - (a.y - o.y) * (b.x - o.x);

        // 快速排斥
        if (Mathf.Max(a.x, b.x) < Mathf.Min(c.x, d.x) ||
            Mathf.Max(c.x, d.x) < Mathf.Min(a.x, b.x) ||
            Mathf.Max(a.y, b.y) < Mathf.Min(c.y, d.y) ||
            Mathf.Max(c.y, d.y) < Mathf.Min(a.y, b.y))
            return false;

        float d1 = Cross(a, b, c);
        float d2 = Cross(a, b, d);
        float d3 = Cross(c, d, a);
        float d4 = Cross(c, d, b);

        if (((d1 > 0 && d2 < 0) || (d1 < 0 && d2 > 0)) &&
            ((d3 > 0 && d4 < 0) || (d3 < 0 && d4 > 0)))
            return true;

        // 共线情况
        if (d1 == 0 && IsPointOnSegment(a, b, c)) return true;
        if (d2 == 0 && IsPointOnSegment(a, b, d)) return true;
        if (d3 == 0 && IsPointOnSegment(c, d, a)) return true;
        if (d4 == 0 && IsPointOnSegment(c, d, b)) return true;

        return false;
    }

    private bool IsPointOnSegment(Vector2 a, Vector2 b, Vector2 c)
    {
        return c.x >= Mathf.Min(a.x, b.x) && c.x <= Mathf.Max(a.x, b.x) &&
               c.y >= Mathf.Min(a.y, b.y) && c.y <= Mathf.Max(a.y, b.y);
    }

    private void DFSWalk(
        int current,
        int parent,
        HashSet<int> usedCreases,
        List<int> currentPath,
        Dictionary<int, List<int>> adj,
        List<(List<int>, HashSet<int>)> results,
        int maxDepth)
    {
        if (currentPath.Count > maxDepth) return;

        // 保存当前状态
        results.Add((new List<int>(currentPath), new HashSet<int>(usedCreases)));

        if (!adj.TryGetValue(current, out var creaseList)) return;

        foreach (int cIndex in creaseList)
        {
            if (usedCreases.Contains(cIndex)) continue;

            var crease = creases[cIndex];
            int next = GetOtherVertex(crease, current);

            if (next == parent) continue; // 不立即回头

            usedCreases.Add(cIndex);
            currentPath.Add(next);
            DFSWalk(next, current, usedCreases, currentPath, adj, results, maxDepth);
            currentPath.RemoveAt(currentPath.Count - 1);
            usedCreases.Remove(cIndex);
        }
    }


    private int GetOtherVertex(OrigamiCrease crease, int vertexIndex)
    {
        return (crease.v1 - 1 == vertexIndex) ? crease.v2 - 1 : crease.v1 - 1;
    }

    private void VisualizeFace(List<int> vertexIndices)
    {
        if (vertexIndices == null || vertexIndices.Count < 3) return;
        if (faceSharedMaterial == null) return;

        GameObject faceObj = new GameObject($"Face_{faces.Count}");
        faceObj.transform.parent = transform;

        MeshFilter mf = faceObj.AddComponent<MeshFilter>();
        MeshRenderer mr = faceObj.AddComponent<MeshRenderer>();

        // 创建顶点数组
        Vector3[] verts = new Vector3[vertexIndices.Count];
        for (int i = 0; i < vertexIndices.Count; i++)
        {
            OrigamiVertex ov = vertices[vertexIndices[i]];
            verts[i] = new Vector3(ov.x, ov.y, ov.z);
        }

        // 2D投影用于三角剖分
        Vector2[] verts2D = new Vector2[vertexIndices.Count];
        for (int i = 0; i < vertexIndices.Count; i++)
            verts2D[i] = new Vector2(verts[i].x, verts[i].z);

        // 三角剖分
        Triangulator triangulator = new Triangulator(verts2D);
        int[] indices = triangulator.Triangulate();

        // 降级策略：三角扇
        if (indices == null || indices.Length == 0)
        {
            Debug.LogWarning("三角剖分失败，使用三角扇降级");
            indices = FallbackTriangulation(vertexIndices.Count);
        }

        // 创建网格
        Mesh mesh = new Mesh();
        mesh.vertices = verts;
        mesh.triangles = indices;
        mesh.RecalculateNormals();
        mesh.RecalculateBounds();

        mf.mesh = mesh;
        mr.material = faceSharedMaterial;

        faceObjects.Add(faceObj);
    }

    private int[] FallbackTriangulation(int vertexCount)
    {
        if (vertexCount < 3) return new int[0];

        List<int> indices = new List<int>();
        for (int i = 1; i < vertexCount - 1; i++)
        {
            indices.Add(0);
            indices.Add(i);
            indices.Add(i + 1);
        }
        return indices.ToArray();
    }

    private void MoveSelectedVertex(Vector3 position)
    {
        if (selectedVertexIndex < 0) return;

        vertices[selectedVertexIndex].x = position.x;
        vertices[selectedVertexIndex].y = position.y;
        vertices[selectedVertexIndex].z = position.z;
        vertexObjects[selectedVertexIndex].transform.position = position;
        UpdateConnectedCreases(selectedVertexIndex);
    }

    private void UpdateConnectedCreases(int vertexIndex)
    {
        for (int i = 0; i < creases.Count; i++)
        {
            var c = creases[i];
            if (c.v1 == vertexIndex + 1 || c.v2 == vertexIndex + 1)
            {
                if (i >= creaseObjects.Count || creaseObjects[i] == null) continue;

                var lr = creaseObjects[i].GetComponent<LineRenderer>();
                if (lr == null) continue;

                int v1 = c.v1 - 1;
                int v2 = c.v2 - 1;
                lr.SetPosition(0, new Vector3(vertices[v1].x, vertices[v1].y, vertices[v1].z));
                lr.SetPosition(1, new Vector3(vertices[v2].x, vertices[v2].y, vertices[v2].z));
            }
        }
    }

    private int FindNearestVertex(Vector3 position)
    {
        float minDist = snapDistance;
        int nearest = -1;

        // 优化：使用平方距离避免开方
        float minDistSq = snapDistance * snapDistance;
        for (int i = 0; i < vertices.Count; i++)
        {
            Vector3 vPos = new Vector3(vertices[i].x, vertices[i].y, vertices[i].z);
            float dx = position.x - vPos.x;
            float dz = position.z - vPos.z;
            float distSq = dx * dx + dz * dz;

            if (distSq < minDistSq)
            {
                minDistSq = distSq;
                nearest = i;
            }
        }
        return nearest;
    }

    private void CancelCurrentOperation()
    {
        if (startVertexIndex >= 0 && startVertexIndex < vertexObjects.Count)
        {
            var renderer = vertexObjects[startVertexIndex]?.GetComponent<Renderer>();
            if (renderer != null && vertexMaterial != null)
                renderer.material = vertexMaterial;
        }

        startVertexIndex = -1;
        selectedVertexIndex = -1;
    }

    // 三角剖分类
    public class Triangulator
    {
        private List<Vector2> m_points;

        public Triangulator(Vector2[] points)
        {
            m_points = new List<Vector2>(points);
        }

        public int[] Triangulate()
        {
            List<int> indices = new List<int>();
            int n = m_points.Count;
            if (n < 3) return indices.ToArray();

            // 处理顶点顺序
            int[] V = new int[n];
            if (Area() > 0)
            {
                for (int v = 0; v < n; v++) V[v] = v;
            }
            else
            {
                for (int v = 0; v < n; v++) V[v] = (n - 1) - v;
            }

            int nv = n;
            int count = 2 * nv;
            for (int v = nv - 1; nv > 2;)
            {
                if ((count--) <= 0) return indices.ToArray(); // 防止死循环

                int u = v; if (nv <= u) u = 0;
                v = u + 1; if (nv <= v) v = 0;
                int w = v + 1; if (nv <= w) w = 0;

                if (Snip(u, v, w, nv, V))
                {
                    int a = V[u], b = V[v], c = V[w];
                    indices.Add(a);
                    indices.Add(b);
                    indices.Add(c);

                    // 移除已处理顶点
                    for (int s = v, t = v + 1; t < nv; s++, t++)
                        V[s] = V[t];
                    nv--;
                    count = 2 * nv;
                }
            }

            return indices.ToArray();
        }

        private float Area()
        {
            int n = m_points.Count;
            float A = 0.0f;
            for (int p = n - 1, q = 0; q < n; p = q++)
            {
                Vector2 pval = m_points[p];
                Vector2 qval = m_points[q];
                A += pval.x * qval.y - qval.x * pval.y;
            }
            return A * 0.5f;
        }

        private bool Snip(int u, int v, int w, int n, int[] V)
        {
            Vector2 A = m_points[V[u]];
            Vector2 B = m_points[V[v]];
            Vector2 C = m_points[V[w]];

            // 检查是否为有效三角形
            float cross = ((B.x - A.x) * (C.y - A.y)) - ((B.y - A.y) * (C.x - A.x));
            if (cross < Mathf.Epsilon) return false;

            // 检查内部点
            for (int p = 0; p < n; p++)
            {
                if (p == u || p == v || p == w) continue;
                Vector2 P = m_points[V[p]];
                if (InsideTriangle(A, B, C, P)) return false;
            }
            return true;
        }

        private bool InsideTriangle(Vector2 A, Vector2 B, Vector2 C, Vector2 P)
        {
            float ax = C.x - B.x, ay = C.y - B.y;
            float bx = A.x - C.x, by = A.y - C.y;
            float cx = B.x - A.x, cy = B.y - A.y;
            float apx = P.x - A.x, apy = P.y - A.y;
            float bpx = P.x - B.x, bpy = P.y - B.y;
            float cpx = P.x - C.x, cpy = P.y - C.y;

            float aCROSSbp = ax * bpy - ay * bpx;
            float cCROSSap = cx * apy - cy * apx;
            float bCROSScp = bx * cpy - by * cpx;

            return (aCROSSbp >= 0f) && (bCROSScp >= 0f) && (cCROSSap >= 0f);
        }
    }
